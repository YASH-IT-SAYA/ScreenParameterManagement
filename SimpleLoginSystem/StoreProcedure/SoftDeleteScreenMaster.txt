ALTER PROCEDURE [dbo].[SoftDeleteScreenMaster]
    @ID INT,
    @IsDeleted BIT,
	@IsActive BIT
AS
BEGIN
UPDATE ScreenMaster
SET IsDeleted = @IsDeleted,IsActive = @IsActive
WHERE ID = @ID;
IF @IsDeleted = 1
	BEGIN
		UPDATE GroupScreenMapping
		SET IsDeleted = 1,IsActive =0 
	WHERE ScreenMasterId = @ID;
	END;
END;
