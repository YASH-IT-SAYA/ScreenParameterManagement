ALTER PROCEDURE [dbo].[UpdateScreenParameterMaster]
    @ID INT,
    @ParameterName NVARCHAR(255),
    @IsActive BIT,
    @IsDeleted BIT
AS
BEGIN
    UPDATE ScreenParameterMaster
    SET Name = @ParameterName, 
        IsActive = @IsActive, 
        IsDeleted = @IsDeleted
    WHERE ID = @ID;
	IF @IsDeleted = 0
	BEGIN
	 UPDATE GroupScreenParameterMapping
        SET IsDeleted = 0,IsActive =1
        WHERE ScreenParameterMasterId = @ID;
	END;
END;
