ALTER PROCEDURE [dbo].[AddScreenParameter]
    @ScreenMasterId INT,
    @Name NVARCHAR(255),
    @IsDeleted BIT,
    @IsActive BIT
AS
BEGIN
    declare @NewScreenParameterMasterId int;
    INSERT INTO ScreenParameterMaster (ScreenMasterId, Name, IsDeleted, IsActive)
    VALUES (@ScreenMasterId, @Name, @IsDeleted, @IsActive);

	set @NewScreenParameterMasterId = SCOPE_IDENTITY();
 
        INSERT INTO GroupScreenParameterMapping(
        ScreenParameterMasterId,
        GroupId,
        IsSelected,
        CreatedBy,
        IsActive,
        IsDeleted,
        CreatedOn
    )
    SELECT
        @NewScreenParameterMasterId,
        GD.Id,
        0,              
        1,           
        1,             
        0,              
        GETDATE()       
    FROM GroupDetail GD
    WHERE GD.IsDeleted = 0;
END;
