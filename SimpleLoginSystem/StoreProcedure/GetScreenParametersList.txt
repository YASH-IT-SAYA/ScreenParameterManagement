ALTER PROCEDURE [dbo].[GetScreenParametersList]
   @ScreenMasterId INT
AS
BEGIN
   SELECT ID AS ParameterID,
          Name AS ParameterName,
          IsActive,
          IsDeleted
   FROM ScreenParameterMaster
   WHERE ScreenMasterId = @ScreenMasterId 
END
