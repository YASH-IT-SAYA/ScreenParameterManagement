ALTER PROCEDURE [dbo].[SoftDeleteScreenParameter]
    @ID INT,
    @IsDeleted BIT,
	@IsActive BIT
AS
BEGIN
    UPDATE ScreenParameterMaster
    SET IsDeleted = @IsDeleted,IsActive = @IsActive
    WHERE ID = @ID;
	IF @IsDeleted = 1
	BEGIN
		UPDATE GroupScreenParameterMapping
		SET IsDeleted = 1,IsActive =0 
	WHERE ScreenParameterMasterId = @ID;
	END;
END;
