ALTER PROCEDURE [dbo].[UpdateScreenMaster]
    @ID INT,
    @Name NVARCHAR(255),
    @IsActive BIT,
    @IsDeleted BIT
AS
BEGIN
    UPDATE ScreenMaster
    SET Name = @Name, 
        IsActive = @IsActive, 
        IsDeleted = @IsDeleted
    WHERE ID = @ID;
	IF @IsDeleted = 0
	BEGIN
	 UPDATE GroupScreenMapping
        SET IsDeleted = 0,IsActive =1
        WHERE ScreenMasterId = @ID;
	END;
END;
