ALTER PROCEDURE [dbo].[AddScreenMaster]
    @Name NVARCHAR(255),
    @IsActive BIT,
    @IsDeleted BIT
AS
BEGIN
    DECLARE @NewScreenMasterId INT;
    
    INSERT INTO ScreenMaster (Name, IsActive, IsDeleted)
    VALUES (@Name, @IsActive, @IsDeleted);

    SET @NewScreenMasterId = SCOPE_IDENTITY();  

    
    INSERT INTO GroupScreenMapping (
        ScreenMasterId,
        GroupId,
        IsSelected,
        CreatedBy,
        IsActive,
        IsDeleted,
        CreatedOn
    )
    SELECT
        @NewScreenMasterId,
        GD.Id,
        0,              
        1,           
        1,             
        0,              
        GETDATE()       
    FROM GroupDetail GD
    WHERE GD.IsDeleted = 0;
END;
